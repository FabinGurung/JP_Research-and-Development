\begin{thesisabstract}
\phantomsection\label{front:abstract}
\abstractparagraph{This study examines the influence of soil--structure interaction on the G+4 Infectious and Communicable Disease Hospital in Lekhnath, Pokhara. Six reinforced-concrete structural blocks were analysed under fixed-base, hard-, medium- and soft-soil support conditions. Modal response, response-spectrum displacement, inter-storey drift and shear were compared alongside foundation settlement, vertical reaction and member-force demands. X-direction seismic fragility curves were also examined.}

\abstractparagraph{The first-mode period increased from fixed-base to soft-soil support by 8.53\%, 19.73\%, 5.22\%, 8.99\%, 9.00\% and 7.85\% for Blocks 1--6, respectively. At least 90\% cumulative mass participation in both horizontal directions was attained between Modes 5 and 10. X-direction uppermost-storey displacement increased under soft-soil support for all six blocks. The highest maximum inter-storey drift was 0.774\% for Block 3 under soft-soil support. Foundation movements increased with flexibility, whereas shear and member-force changes varied with the response quantity and structural block. The results show that period lengthening alone does not determine seismic force demand; the influence of soil--structure interaction must be assessed using multiple response measures.}

\abstractkeywords{Soil--Structure Interaction; Hospital Buildings; Seismic Response; Foundation Flexibility; Response-Spectrum Analysis; Seismic Fragility.}
\end{thesisabstract}
